"""One bounded PG design probe; not a product worker, migration or recovery API."""

import hashlib
import json
from pathlib import Path
from uuid import uuid4

import psycopg
from psycopg.types.json import Jsonb

from scripts.m0.postgres_lab import DATA, DSN, start, stop, verify_server

ROOT = Path(__file__).resolve().parents[1]
SCHEMA = "m0_vertical_probe_20260909"


def sha(value):
    return hashlib.sha256(value).hexdigest()


def connect():
    return psycopg.connect(
        DSN, connect_timeout=2, options="-c statement_timeout=5000 -c lock_timeout=2000"
    )


def live_history(conn):
    return {
        table: sha(
            json.dumps(
                conn.execute(
                    f"SELECT row_to_json(t) FROM {table} t ORDER BY experiment_id"
                ).fetchall(),
                sort_keys=True,
                default=str,
            ).encode()
        )
        for table in ("m0_live_once", "m0_live_diagnostics")
    }


def adopt(conn, subject, generation, epoch, final):
    # Single SQL predicate checks the current durable authority at adoption time.
    accepted = conn.execute(
        f"UPDATE {SCHEMA}.subject SET status='completed',final=%s "
        "WHERE id=%s AND generation=%s AND epoch=%s AND status='running' "
        "AND lease_until>clock_timestamp() RETURNING id",
        (Jsonb(final), subject, generation, epoch),
    ).fetchone()
    conn.execute(
        f"INSERT INTO {SCHEMA}.attempt VALUES (%s,%s,%s,%s,%s)",
        (uuid4(), subject, generation, epoch, bool(accepted)),
    )
    return bool(accepted)


def main():
    verify_server()
    artifact = ROOT / "docs/evidence/m0-real-investigation/flash-results.json"
    live = json.loads(artifact.read_text())[1]
    final_path = ROOT / "docs/evidence/m0-real-investigation/final-diagnostic-2.json"
    final = json.loads(
        json.loads(final_path.read_text())["known_fixture_final_content"]
    )
    assert live["result"]["business"] == "completed"
    snapshot = {
        "schema": "m0-vertical-probe-v1",
        "source_experiment": live["row"]["experiment_id"],
        "source_run": live["row"]["run_id"],
        "source_code": live["version"]["code_sha256"],
        "model": "deepseek-v4-flash",
        "fixture": "m0-protocol-v1",
        "final": final,
        "evidence": json.loads(
            (ROOT / "tests/fixtures/m0/protocol-v1.json").read_text()
        )["evidence"],
    }
    snapshot_hash = sha(json.dumps(snapshot, sort_keys=True).encode())
    subjects = {name: uuid4() for name in ("normal", "cancel", "epoch", "expired")}
    with connect() as conn:
        before = live_history(conn)
        # No IF NOT EXISTS and no DROP: an existing probe is preserved and refuses rerun.
        conn.execute(f"CREATE SCHEMA {SCHEMA}")
        conn.execute(
            f"CREATE TABLE {SCHEMA}.subject (id uuid PRIMARY KEY, status text NOT NULL, "
            "generation int NOT NULL, epoch int NOT NULL, lease_until timestamptz NOT NULL, "
            "snapshot jsonb NOT NULL, snapshot_sha text NOT NULL, final jsonb)"
        )
        conn.execute(
            f"CREATE TABLE {SCHEMA}.attempt (id uuid PRIMARY KEY, subject uuid NOT NULL "
            f"REFERENCES {SCHEMA}.subject(id), generation int, epoch int, accepted bool)"
        )
        for subject in subjects.values():
            conn.execute(
                f"INSERT INTO {SCHEMA}.subject VALUES "
                "(%s,'running',0,1,clock_timestamp()+interval '15 minutes',%s,%s,NULL)",
                (subject, Jsonb(snapshot), snapshot_hash),
            )
    with connect() as conn:
        assert adopt(conn, subjects["normal"], 0, 1, final)
    with connect() as conn:
        conn.execute(
            f"UPDATE {SCHEMA}.subject SET status='cancelled',generation=generation+1 WHERE id=%s",
            (subjects["cancel"],),
        )
        conn.execute(
            f"UPDATE {SCHEMA}.subject SET epoch=2 WHERE id=%s", (subjects["epoch"],)
        )
        conn.execute(
            f"UPDATE {SCHEMA}.subject SET lease_until=clock_timestamp()-interval '1 second' WHERE id=%s",
            (subjects["expired"],),
        )
    # A new connection acts as the late worker after human control has committed.
    with connect() as conn:
        for name in ("cancel", "epoch", "expired"):
            assert not adopt(conn, subjects[name], 0, 1, final)
        assert not adopt(conn, subjects["cancel"], 1, 1, final)
    old_pid = (DATA / "postmaster.pid").read_text().splitlines()[0]
    stop()
    start()
    new_pid = (DATA / "postmaster.pid").read_text().splitlines()[0]
    assert old_pid != new_pid
    with connect() as conn:
        rows = conn.execute(
            f"SELECT id,status,generation,epoch,snapshot,snapshot_sha,final FROM {SCHEMA}.subject ORDER BY id"
        ).fetchall()
        assert len(rows) == 4
        for row in rows:
            assert row[4] == snapshot
            assert sha(json.dumps(row[4], sort_keys=True).encode()) == row[5]
            if row[0] == subjects["cancel"]:
                assert row[1:3] == ("cancelled", 1) and row[6] is None
        attempts = conn.execute(
            f"SELECT accepted,count(*) FROM {SCHEMA}.attempt GROUP BY accepted ORDER BY accepted"
        ).fetchall()
        assert attempts == [(False, 4), (True, 1)]
        assert live_history(conn) == before
        # Explicit reconstruction compatibility check, not checkpoint reuse.
        compatible = all(r[4]["schema"] == "m0-vertical-probe-v1" for r in rows)
        incompatible = any(r[4]["schema"] != "m0-vertical-probe-v2" for r in rows)
        assert compatible and incompatible
    result = {
        "schema": SCHEMA,
        "source_experiment": snapshot["source_experiment"],
        "source_sha256": sha(Path(__file__).read_bytes()),
        "snapshot_sha256": snapshot_hash,
        "normal_adoption": 1,
        "late_cancel_epoch_expiry_rejected": 4,
        "restart_pid_changed": True,
        "snapshots_preserved": 4,
        "live_business_and_diagnostics_unchanged": True,
        "incompatible_reconstruction": "blocked(INCOMPATIBLE_STATE)",
        "model_requests": 0,
        "trace_uploads": 0,
        "boundary": "probe SQL mechanics and persisted known business snapshot only; no product worker or provider-private continuation",
    }
    out = ROOT / "docs/evidence/m0-real-investigation/control-probe.json"
    with out.open("x") as file:
        json.dump(result, file, indent=2)
        file.write("\n")
    print(json.dumps(result))


if __name__ == "__main__":
    main()
